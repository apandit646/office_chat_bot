### Load Document, Chunk and Store in ChromaDB

First, we load the PDF document using `PyPDFLoader`. Then, we split the document into smaller, overlapping chunks using `RecursiveCharacterTextSplitter` to maintain context. Finally, we generate embeddings for these chunks using `GoogleGenerativeAIEmbeddings` and store them in a ChromaDB vector store.

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_google_genai import GoogleGenerativeAIEmbeddings
from langchain_community.vectorstores import Chroma
from google.colab import userdata

# Set your Google API Key
GOOGLE_API_KEY = userdata.get('GOOGLE_API_KEY')

# Load the PDF document
pdf_path = "/content/sample_data/Policy for Reimbursement of Technical Certifications_Ver_1.01.pdf"
loader = PyPDFLoader(pdf_path)
docs = loader.load()

# Split the document into chunks
text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
chunks = text_splitter.split_documents(docs)

# Initialize Google Generative AI Embeddings with the correct model
embeddings = GoogleGenerativeAIEmbeddings(model="models/gemini-embedding-001", google_api_key=GOOGLE_API_KEY)

# Create and store in ChromaDB
vectorstore = Chroma.from_documents(documents=chunks, embedding=embeddings, persist_directory="./chroma_db")

print(f"Successfully loaded {len(docs)} pages, split into {len(chunks)} chunks, and stored in ChromaDB.")

"""### Build a Module-Based Chatbot using LangGraph

Now that we have our document chunks stored in ChromaDB, we can build a RAG (Retrieval Augmented Generation) chatbot using LangGraph. This involves defining the state of our conversation, creating nodes for retrieval and generation, and then connecting them in a graph.
"""

from typing import List
from typing_extensions import TypedDict

from langchain_core.messages import BaseMessage, HumanMessage
from langchain_core.documents import Document

# Define the graph state
class GraphState(TypedDict):
    """Represents the state of our graph.

    Attributes:
        question: question
        conversation_history: list of messages
        documents: list of documents
        generation: LLM generation
    """
    question: str
    conversation_history: List[BaseMessage]
    documents: List[Document]
    generation: str

from langchain_google_genai import ChatGoogleGenerativeAI
from langchain_core.runnables import RunnablePassthrough
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder # Added MessagesPlaceholder

# Initialize the LLM for generation
llm = ChatGoogleGenerativeAI(model="gemini-2.5-flash", google_api_key=GOOGLE_API_KEY, temperature=0)

# Prompt for generation - now includes chat_history
GENERATION_PROMPT = ChatPromptTemplate.from_messages([
    ("system", "You are an AI assistant for a document. Use the following pieces of retrieved context to answer the question. If you don't know the answer, just say that you don't know."),
    MessagesPlaceholder(variable_name="chat_history"), # Placeholder for previous conversation
    ("human", "Question: {question}\nContext: {context}\nAnswer:"),
])

# Chain for generation
generation_chain = GENERATION_PROMPT | llm | StrOutputParser()

def retrieve(state: GraphState):
    """Retrieves documents from the vector store based on the question."""
    print("---RETRIEVE DOCUMENTS---")
    question = state["question"]
    documents = vectorstore.similarity_search(question)
    return {"documents": documents, "question": question}

from langchain_core.messages import AIMessage, HumanMessage # Added imports for AIMessage and HumanMessage

def generate(state: GraphState):
    """Generates an answer using the retrieved documents and the question, and updates conversation history."""
    print("---GENERATE ANSWER---")
    question = state["question"]
    documents = state["documents"]
    conversation_history = state.get("conversation_history", []) # Get history, default to empty list

    # Invoke generation chain with history
    generation_output = generation_chain.invoke({
        "question": question,
        "context": documents,
        "chat_history": conversation_history # Pass the full list of BaseMessage
    })

    # Update conversation history with current turn
    new_conversation_history = conversation_history + [
        HumanMessage(content=question),
        AIMessage(content=generation_output)
    ]

    return {
        "documents": documents,
        "question": question,
        "generation": generation_output,
        "conversation_history": new_conversation_history
    }

from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver # Import MemorySaver

# Build the LangGraph agent
workflow = StateGraph(GraphState)

# Add nodes
workflow.add_node("retrieve", retrieve)  # Retrieve documents
workflow.add_node("generate", generate)  # Generate answer

# Set entry point
workflow.set_entry_point("retrieve")

# Add edges
workflow.add_edge("retrieve", "generate")
workflow.add_edge("generate", END)

# Initialize MemorySaver for checkpointing
memory = MemorySaver()

# Compile the graph with the checkpointer
app = workflow.compile(checkpointer=memory)

"""### Test the Chatbot"""

# Initial inputs for the first turn
inputs = {"question": "What is the policy for technical certifications?", "conversation_history": []}
response = app.invoke(inputs, config={"configurable": {"thread_id": "1"}})
print("First turn:", response["generation"])

# Second turn, using the updated history from the first response
print("\n--- Second Turn ---")
second_question = "Can you elaborate on the eligibility criteria?"
inputs_second_turn = {"question": second_question, "conversation_history": response["conversation_history"]}
response_second_turn = app.invoke(inputs_second_turn, config={"configurable": {"thread_id": "1"}})
print("Second turn:", response_second_turn["generation"])

# Third turn, asking a follow-up question that relies on context from previous turns
print("\n--- Third Turn ---")
third_question = "What about the maximum amount?"
inputs_third_turn = {"question": third_question, "conversation_history": response_second_turn["conversation_history"]}
response_third_turn = app.invoke(inputs_third_turn, config={"configurable": {"thread_id": "1"}})
print("Third turn:", response_third_turn["generation"])
